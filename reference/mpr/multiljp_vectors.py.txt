async def get_embedding_with_retry(text: str, max_retries=3) -> np.ndarray:
    """带重试机制的异步OpenAI嵌入获取"""
    if text.strip() == "":
        return np.zeros(dimension, dtype=np.float32)
    for attempt in range(max_retries):
        try:
            response = embeddingModel.embeddings.create(
                input=text,
                model="text-embedding-ada-002"
            )
            return np.array(response.data[0].embedding, dtype=np.float32)
        except Exception as e:
            if attempt < max_retries - 1:
                wait = 2 ** attempt  # 指数退避
                print(f"API错误：{str(e)}，{wait}秒后重试...")
                await asyncio.sleep(wait)
            else:
                print(f"API失败：{text[:50]}...")
                return np.zeros(dimension, dtype=np.float32)
    return np.zeros(dimension, dtype=np.float32)


async def process_case_data(dic):

    vectors = {}

    tasks = [
        get_embedding_with_retry(dic['data']['elements']["犯罪事实"]),
        get_embedding_with_retry(dic['data']['elements']["行为"]),
        get_embedding_with_retry(dic['data']['elements']["后果"]),
    ]

    results = await asyncio.gather(*tasks)

    vectors["fact"] = results[0]
    vectors["behavior"] = results[1]
    vectors["consequence"] = results[2]

    data_1 = {
        "origin_data": dic,
        "fact_emb": vectors["fact"],
        "behavior": vectors["behavior"],
        "consequence": vectors["consequence"],
    }

    save_json_data(data_1)


def save_json_data(json_data, file_path="test_data_with_embedding_1.jsonl"):
    json_data['fact_emb'] = json_data['fact_emb'].tolist()
    json_data['behavior'] = json_data['behavior'].tolist()
    json_data['consequence'] = json_data['consequence'].tolist()


    with open(file_path, "a", encoding="utf-8") as f:
        json.dump(json_data, f, ensure_ascii=False)
        f.write('\n')


def get_processed_ids(file_path="test_data_with_embedding_1.jsonl"):
    processed_ids = set()
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            for line in f:
                try:
                    dic_data = json.loads(line)
                    if isinstance(dic_data, dict):
                        processed_ids.add(dic_data.get("case_id", None))
                    else:
                        print(f"Line is not a dictionary: {line}")
                except json.JSONDecodeError as e:
                    print(f"Error decoding line in {file_path}: {e}")
                    continue
    except FileNotFoundError:
        print(f"File not found: {file_path}")
    return processed_ids


async def process_case_data_async(dic):
    await process_case_data(dic)
