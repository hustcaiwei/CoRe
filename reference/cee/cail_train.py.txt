import json
import asyncio
import requests
from tqdm import tqdm
import time  # 引入时间模块来记录花费的时间

from tenacity import retry, stop_after_attempt, wait_fixed, RetryError

# 设置请求超时时间为 10 秒，超过此时间未响应则视为超时
TIMEOUT = 30

def get_processed_ids(file_path="train_test_1.jsonl"):
    processed_ids = set()
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            for line in f:
                try:
                    data = json.loads(line)
                    if isinstance(data, dict):  # Ensure the data is a dictionary
                        processed_ids.add(data.get("id", None))  # Using .get() in case "id" doesn't exist
                    else:
                        print(f"Line is not a dictionary: {line}")
                except json.JSONDecodeError as e:
                    print(f"Error decoding line in {file_path}: {e}")
                    continue
    except FileNotFoundError:
        print(f"File not found: {file_path}")
    return processed_ids

def save_json_data(all_json_data, file_path="train_test_1.jsonl"):
    with open(file_path, "a", encoding="utf-8") as f:
        for json_data in all_json_data:
            json.dump(json_data, f, ensure_ascii=False)
            f.write('\n')


@retry(stop=stop_after_attempt(3), wait=wait_fixed(2))  # 重试最多3次，每次间隔2秒
def deepseek_query(case, id, accu, all_json_data):
    url = "https://dashscope.aliyuncs.com/compatible-mode/v1/chat/completions"
    headers = {
        "Authorization": "Bearer <API_KEY_REMOVED>",  # API key
        "Content-Type": "application/json"
    }

    prompt = """
        "任务说明": "根据提供的刑事案例文本和罪名，精准提取定罪依据、犯罪后果及量刑情节，输出结构化JSON数据。",
        "输入要求": {
        "案例文本": "需分析的原始案例文本",
                "罪名": "刑法明确规定的罪名（如盗窃罪、故意伤害罪等）"
                },
                "处理规则": [
                "1. **定罪依据**：",
                "- 严格匹配罪名构成要件，仅提取直接支撑该罪名的犯罪行为描述",
                "- 删除时间、地点、人物背景等无关信息",
                "- 使用最简短的主动语态（如'持刀砍伤被害人'而非'被害人被其持刀砍伤'）",
                "2. **犯罪后果**：",
                "- 物质损失：量化表述（如'损失5000元'）",
                "- 人身伤害：采用司法鉴定术语（如'轻伤二级'）",
                "- 无具体后果则标注'无'",
                "3. **量刑情节**：",
                "- 第1层面：仅限刑法总则规定的16种法定情节（如未遂、从犯等）",
                "- 第2层面：仅限司法解释规定的酌定情节（如自首、赔偿等）",
                "- 存在多种情节时分行列举（换行符\\n分隔）",
                "- 绝对避免使用'或''可能'等模糊表述",
                "4. **罪名区分提示**：",
                "- 寻衅滋事罪：必须存在'无事生非''逞强耍横'动机描述",
                "- 聚众斗殴罪：需出现'纠集三人以上'等关键词",
                "- 职务犯罪：必须明确'利用职务便利'行为"
                ],
                "输出规范": {
                "格式": "严格遵循以下JSON结构，值部分必须用双引号",
                "示例": {
                "定罪依据": "持刀刺穿被害人腹部",
                "犯罪后果": "重伤二级",
                "第1层面量刑情节": "犯罪未遂",
                "第2层面量刑情节": "自首\\n赔偿被害人损失"
                }
                },
                "特殊指令": [
                "如案例文本出现以下干扰信息必须删除：",
                "- 地名（如'在北京市朝阳区'）",
                "- 当事人称谓（如'被告人张某'改为'行为人'）",
                "- 判决结果（如'判处有期徒刑三年'）",
                "- 情感描述（如'恶劣''残忍'等主观评价）"
                ]
        """

    # 替换案例文本
    prompt = prompt.replace("{{case_text}}", case)

    # 请求体
    data_response = {
        "model": "deepseek-v3",
        "messages": [
            {"role": "system", "content": prompt},
            {"role": "user", "content": case}
        ],
        "temperature": 0.3,
        "stream": False,
        "response_format": {
            "type": "json_object"
        }
    }

    try:
        # 发送请求并添加超时设置
        response = requests.post(url, json=data_response, headers=headers, timeout=TIMEOUT)

        if response.status_code == 200:
            response_json = response.json()
            elements_content = response_json['choices'][0]['message']['content']
            print(id)
            print(accu)
            print(elements_content)
            json_str = elements_content.strip().removeprefix('```json').removesuffix('```').strip()
            data_dict = json.loads(json_str)
            json_data = {
                "id": id,
                "behavior": data_dict['定罪依据'],
                "consequence": data_dict['犯罪后果'],
                "first_level_violations": data_dict['第1层面量刑情节'],
                "second_level_violations": data_dict['第2层面量刑情节']
            }

            all_json_data.append(json_data)
            return json_data
        else:
            print(f"Failed request with status code: {response.status_code}")
            return None
    except requests.exceptions.Timeout as e:
        print(f"Request timed out: {e}")
        raise  # 触发重试
    except requests.exceptions.RequestException as e:
        print(f"Request failed: {e}")
        return None

async def async_deepseek_query(case, id, accu, all_json_data):
    loop = asyncio.get_event_loop()
    return await loop.run_in_executor(None, deepseek_query, case, id, accu, all_json_data)

# 异步读取和处理文件
async def main():
    processed_ids = get_processed_ids()  # 获取已处理的 ID 列表
    print(f"已读取 {len(processed_ids)} 个已处理的案例 ID")

    all_json_data = []  # 用于保存所有返回的数据
    path = '<LOCAL_PATH>/new_dataset.jsonl'
    batch_size = 30  # 每批处理 10 个案例
    batch_count = 0
    total_lines = sum(1 for line in open(path, "r", encoding="utf-8"))  # 计算文件总行数
    tasks = []  # 异步请求任务列表
    batch_start_time = time.time()
    task_count = 0  # 跟踪已处理的任务数

    with open(path, "r", encoding="utf-8") as f:
        pbar = tqdm(f, desc="犯罪案例总结中", total=total_lines)  # 进度条显示总行数
        for i, line in enumerate(pbar):
            data = json.loads(line)
            if data["id"] in processed_ids:  # 跳过已处理的
                continue
            case = data["fact_cut"]
            accu = accu_content[data['accu']]
            term = data["term"]
            id = data['id']

            # 异步请求 DeepSeek API
            tasks.append(async_deepseek_query(case, id, accu, all_json_data))
            task_count += 1  # 增加任务数

            if len(tasks) >= batch_size:  # 当任务数量达到批处理大小时，开始处理
                results = await asyncio.gather(*tasks)  # 等待所有任务完成
                tasks.clear()  # 清空任务列表

                batch_end_time = time.time()
                batch_duration = batch_end_time - batch_start_time
                print(f"第 {batch_count + 1} 批处理完毕，总耗时 {batch_duration:.2f} 秒")
                batch_start_time = time.time()  # 重新开始计时
                batch_count += 1  # 增加批次计数

            if task_count >= 300:  # 每处理 1000 个任务就保存一次
                save_json_data(all_json_data)
                all_json_data.clear()  # 清空数据列表
                task_count = 0  # 重置任务数

        if tasks:  # 处理剩余的任务
            results = await asyncio.gather(*tasks)

            batch_end_time = time.time()
            batch_duration = batch_end_time - batch_start_time
            print(f"第 {batch_count + 1} 批处理完毕，总耗时 {batch_duration:.2f} 秒")

            # 保存剩余的数据
        if all_json_data:
            save_json_data(all_json_data)
