import re
from collections import defaultdict
from sklearn.metrics import f1_score, precision_score, recall_score, accuracy_score
import numpy as np


def parse_log_file(log_file_path):
    """
    解析日志文件，提取预测和真实值
    """
    results = []

    with open(log_file_path, 'r', encoding='gbk') as f:
        for line in f:
            # 匹配被告预测和真实值的行
            match = re.search(r"预测:\s*\((\d+),\s*'([^']+)',\s*(\d+)\)\s*\|\s*真实:\s*\((\d+),\s*'([^']+)',\s*(\d+)\)",
                              line)
            if match:
                defendant = match.group(1)
                predicted_law = int(match.group(1))  # 预测的法条
                predicted_crime = match.group(2).strip()  # 预测的罪名
                predicted_term = int(match.group(3))  # 预测的刑期

                true_law = int(match.group(4))  # 真实的法条
                true_crime = match.group(5).strip()  # 真实的罪名
                true_term = int(match.group(6))  # 真实的刑期

                # 处理乱码情况
                if "预测值解析失败" in predicted_crime or "错误:" in predicted_crime:
                    continue

                results.append({
                    'defendant': defendant,
                    'pred': (predicted_law, predicted_crime, predicted_term),
                    'true': (true_law, true_crime, true_term)
                })

    return results


def evaluate(predictions, labels):
    """
    计算评估指标
    """
    # 提取各个任务的预测值和真实值
    y_true_law = [label[0] for label in labels]
    y_pred_law = [pred[0] for pred in predictions]

    y_true_crime = [label[1] for label in labels]
    y_pred_crime = [pred[1] for pred in predictions]

    y_true_term = [label[2] for label in labels]
    y_pred_term = [pred[2] for pred in predictions]

    # 获取所有唯一类别
    all_law_classes = sorted(set(y_true_law + y_pred_law))
    all_crime_classes = sorted(set(y_true_crime + y_pred_crime))
    all_term_classes = sorted(set(y_true_term + y_pred_term))

    # 计算法条指标
    law_metrics = calculate_metrics_per_class(y_true_law, y_pred_law, all_law_classes, "法条")

    # 计算罪名指标
    crime_metrics = calculate_metrics_per_class(y_true_crime, y_pred_crime, all_crime_classes, "罪名")

    # 计算刑期指标
    term_metrics = calculate_metrics_per_class(y_true_term, y_pred_term, all_term_classes, "刑期")

    # 打印详细结果
    print_detailed_metrics(law_metrics, "法条")
    print_detailed_metrics(crime_metrics, "罪名")
    print_detailed_metrics(term_metrics, "刑期")

    # 计算宏平均和微平均
    law_macro_f1 = f1_score(y_true_law, y_pred_law, average='macro', zero_division=0)
    law_macro_precision = precision_score(y_true_law, y_pred_law, average='macro', zero_division=0)
    law_macro_recall = recall_score(y_true_law, y_pred_law, average='macro', zero_division=0)
    law_micro_f1 = f1_score(y_true_law, y_pred_law, average='micro', zero_division=0)
    law_micro_precision = precision_score(y_true_law, y_pred_law, average='micro', zero_division=0)
    law_micro_recall = recall_score(y_true_law, y_pred_law, average='micro', zero_division=0)
    law_accuracy = accuracy_score(y_true_law, y_pred_law)

    crime_macro_f1 = f1_score(y_true_crime, y_pred_crime, average='macro', zero_division=0)
    crime_macro_precision = precision_score(y_true_crime, y_pred_crime, average='macro', zero_division=0)
    crime_macro_recall = recall_score(y_true_crime, y_pred_crime, average='macro', zero_division=0)
    crime_micro_f1 = f1_score(y_true_crime, y_pred_crime, average='micro', zero_division=0)
    crime_micro_precision = precision_score(y_true_crime, y_pred_crime, average='micro', zero_division=0)
    crime_micro_recall = recall_score(y_true_crime, y_pred_crime, average='micro', zero_division=0)
    crime_accuracy = accuracy_score(y_true_crime, y_pred_crime)

    term_macro_f1 = f1_score(y_true_term, y_pred_term, average='macro', zero_division=0)
    term_macro_precision = precision_score(y_true_term, y_pred_term, average='macro', zero_division=0)
    term_macro_recall = recall_score(y_true_term, y_pred_term, average='macro', zero_division=0)
    term_micro_f1 = f1_score(y_true_term, y_pred_term, average='micro', zero_division=0)
    term_micro_precision = precision_score(y_true_term, y_pred_term, average='micro', zero_division=0)
    term_micro_recall = recall_score(y_true_term, y_pred_term, average='micro', zero_division=0)
    term_accuracy = accuracy_score(y_true_term, y_pred_term)

    # 打印汇总统计
    print_summary_stats("法条", law_accuracy, law_macro_f1, law_macro_precision, law_macro_recall,
                        law_micro_f1, law_micro_precision, law_micro_recall)
    print_summary_stats("罪名", crime_accuracy, crime_macro_f1, crime_macro_precision, crime_macro_recall,
                        crime_micro_f1, crime_micro_precision, crime_micro_recall)
    print_summary_stats("刑期", term_accuracy, term_macro_f1, term_macro_precision, term_macro_recall,
                        term_micro_f1, term_micro_precision, term_micro_recall)

    return {
        'law': {
            'macro': {
                'precision': law_macro_precision,
                'recall': law_macro_recall,
                'f1': law_macro_f1
            },
            'micro': {
                'precision': law_micro_precision,
                'recall': law_micro_recall,
                'f1': law_micro_f1
            },
            'accuracy': law_accuracy,
            'per_class': law_metrics
        },
        'crime': {
            'macro': {
                'precision': crime_macro_precision,
                'recall': crime_macro_recall,
                'f1': crime_macro_f1
            },
            'micro': {
                'precision': crime_micro_precision,
                'recall': crime_micro_recall,
                'f1': crime_micro_f1
            },
            'accuracy': crime_accuracy,
            'per_class': crime_metrics
        },
        'term': {
            'macro': {
                'precision': term_macro_precision,
                'recall': term_macro_recall,
                'f1': term_macro_f1
            },
            'micro': {
                'precision': term_micro_precision,
                'recall': term_micro_recall,
                'f1': term_micro_f1
            },
            'accuracy': term_accuracy,
            'per_class': term_metrics
        }
    }


def calculate_metrics_per_class(y_true, y_pred, all_classes, task_name):
    """
    计算每个类别的指标
    """
    metrics = {}

    for class_label in all_classes:
        # 计算该类别真实样本数
        true_count = sum(1 for x in y_true if x == class_label)

        # 计算该类别预测正确的样本数
        correct_count = sum(1 for yt, yp in zip(y_true, y_pred)
                            if yt == class_label and yp == class_label)

        # 类别准确率 = 预测正确的样本数 / 真实样本数
        class_accuracy = correct_count / true_count if true_count > 0 else 0

        # 创建二分类标签
        y_true_binary = [1 if x == class_label else 0 for x in y_true]
        y_pred_binary = [1 if x == class_label else 0 for x in y_pred]

        precision = precision_score(y_true_binary, y_pred_binary, zero_division=0)
        recall = recall_score(y_true_binary, y_pred_binary, zero_division=0)
        f1 = f1_score(y_true_binary, y_pred_binary, zero_division=0)

        metrics[class_label] = {
            'accuracy': class_accuracy,
            'precision': precision,
            'recall': recall,
            'f1': f1,
            'support': true_count
        }

    return metrics


def print_detailed_metrics(metrics, task_name):
    """
    打印详细类别指标
    """
    print(f"\n=== {task_name}详细类别指标 ===")
    for class_label, metric in sorted(metrics.items()):
        print(f"类别 {class_label} - "
              f"准确率: {metric['accuracy']:.4f}, "
              f"F1: {metric['f1']:.4f}, "
              f"精确率: {metric['precision']:.4f}, "
              f"召回率: {metric['recall']:.4f}, "
              f"支持数: {metric['support']}")


def print_summary_stats(task_name, accuracy, macro_f1, macro_precision, macro_recall,
                        micro_f1, micro_precision, micro_recall):
    """
    打印汇总统计信息
    """
    print(f"\n=== {task_name}汇总统计 ===")
    print(f"  准确率: {accuracy:.4f}")
    print(f"  宏平均 F1: {macro_f1:.4f}")
    print(f"  宏平均 精确率: {macro_precision:.4f}")
    print(f"  宏平均 召回率: {macro_recall:.4f}")
    print(f"  微平均 F1: {micro_f1:.4f}")
    print(f"  微平均 精确率: {micro_precision:.4f}")
    print(f"  微平均 召回率: {micro_recall:.4f}")


def main():
    """
    主函数
    """
    log_file_path = '<LOCAL_PATH>/PlanB_0825_judge_B.log'

    try:
        # 解析日志文件
        parsed_data = parse_log_file(log_file_path)
        print(f"解析到 {len(parsed_data)} 条有效记录")

        if len(parsed_data) == 0:
            print("未找到有效数据，请检查日志文件格式")
            return

        # 准备评估数据
        predictions = [data['pred'] for data in parsed_data]
        labels = [data['true'] for data in parsed_data]

        # 评估结果
        evaluation_results = evaluate(predictions, labels)

    except FileNotFoundError:
        print(f"错误: 找不到日志文件 {log_file_path}")
    except Exception as e:
        print(f"处理日志文件时出错: {e}")


if __name__ == "__main__":
    main()
